# SW‑420

Подключите их к ESP32 следующим образом:
- VCC датчика → 3.3 V или 5 V ESP32 (датчик работает в диапазоне 3.3–5 V);
- GND датчика → GND ESP32;
- DO датчика → любой цифровой вход ESP32 (например, GPIO 15).

```c
  const int vibrationPin = 15;  // Пин, к которому подключён DO датчика

  void setup() {
    Serial.begin(115200);           // Инициализация Serial-порта (скорость 115200 бод)
    pinMode(vibrationPin, INPUT);  // Настройка пина как входа
  }

  void loop() {
    int sensorState = digitalRead(vibrationPin);  // Считываем состояние датчика

    if (sensorState == HIGH) {
      Serial.println("Vibration detected!");  // Вибрация обнаружена
    } else {
      Serial.println("No vibration.");       // Вибрации нет
    }

    delay(100);  // Задержка 100 мс между измерениями
  }
```

## Через Prometheus и Grafana
```c
  #include "config.h"
  #include "certificates.h"
  #include <PromLokiTransport.h>
  #include <PrometheusArduino.h>

  PromLokiTransport transport;
  PromClient client(transport);

  void loop() {
    // Сбор данных с датчиков
    int value = analogRead(A0);

    // Формирование запроса
    WriteRequest req(1, 1024);
    req.addMetric("sensor_value", value);

    // Отправка данных
    client.send(req);

    delay(1000);
  }
```

## Через MQTT и InfluxDB

```c
  #include <ESP8266WiFi.h>
  #include <PubSubClient.h>

  const char* mqttServer = "mqtt.broker.com";
  const int mqttPort = 1883;
  PubSubClient client(mqttServer, mqttPort);

  void loop() {
    float temp = readTemperature(); // Функция чтения температуры
    String message = String("temperature,location=room temp=") + String(temp);
    client.publish("sensors/temperature", message.c_str());
    delay(1000);
  }
```
