#include <WiFi.h>
#include <PubSubClient.h>

// === НАСТРОЙКИ WiFi ===
const char* ssid     = "Ent";        // Замените на свой SSID
const char* password = "Ltzvttjj01";      // Замените на пароль

// === НАСТРОЙКИ MQTT ===
const char* mqtt_server = "192.168.50.182";  // IP сервера с Mosquitto/Home Assistant
const int   mqtt_port   = 1883;   
const char* mqtt_user   = "ntlstl";   // <-- Логин для MQTT (замените!)
const char* mqtt_pass   = "ZkbDkU4zgAmRWzp"; // <-- Пароль для MQTT (замените!)            // Порт MQTT (обычно 1883)
const char* client_id   = "esp32-adc-sensor";   // Уникальный ID клиента


// === НАСТРОЙКИ АЦП ===
const int sampleWindow = 1000;     // Окно сбора данных (мс)
const int inputPin     = 32;     // GPIO 34 (АЦП ESP32)
const int sampleCount  = 50;

WiFiClient wifiClient;
PubSubClient client(wifiClient);


// === ФУНКЦИИ ===
void setup_wifi() {
  Serial.println("\nConnecting to WiFi...");
  WiFi.begin(ssid, password);
  
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  
  Serial.println("\nWiFi connected!");
  Serial.print("IP: ");
  Serial.println(WiFi.localIP());
}

void reconnect() {
  while (!client.connected()) {
    Serial.print("Connecting to MQTT...");
    // <-- Подключение с логином и паролем
    if (client.connect(client_id, mqtt_user, mqtt_pass)) {
      Serial.println(" connected!");
    } else {
      Serial.print("failed, rc=");
      Serial.print(client.state());
      Serial.println(" retrying in 5s...");
      delay(5000);
    }
  }
}

void setup() {
  Serial.begin(115200);
  setup_wifi();
  client.setServer(mqtt_server, mqtt_port);
}

void loop() {
  static unsigned long lastSendTime = 0;
  const unsigned long sendInterval = 1000;  // Отправка раз в 1 с (можно изменить)

  // Переподключение при потере связи
  if (!client.connected()) reconnect();
  client.loop();

  unsigned long currentTime = millis();

  // Если прошло sendInterval мс с последней отправки — собираем и отправляем данные
  if (currentTime - lastSendTime >= sendInterval) {
    lastSendTime = currentTime;

    unsigned long startMillis = millis();  // Начало окна выборки
    unsigned int inMax = 0;              // Максимальное значение
    unsigned int inMin = 4095;         // Минимальное значение (для 12‑битного АЦП)

    // Собираем данные в течение sampleWindow мс
    while (millis() - startMillis < sampleWindow) {
      int adcValue = analogRead(inputPin);  // Считываем значение АЦП

      // Фильтрация аномальных значений
      if (adcValue < 4096) {
        if (adcValue > inMax) {
          inMax = adcValue;  // Обновляем максимум
        }
        if (adcValue < inMin) {
          inMin = adcValue;  // Обновляем минимум
        }
      }
    }

    // Вычисляем амплитуду «пик‑пик»
    unsigned int peakValue = inMax - inMin;

    // Фильтруем шум: если амплитуда < 20, считаем её нулевой
    if (peakValue < 20) {
      peakValue = 0;
    }

    // Подготовка строки для MQTT
    char peakStr[10];
    sprintf(peakStr, "%u", peakValue);

    // Отправка в MQTT
    client.publish("sensor/adc/peak", peakStr);

    // Опциональный лог (можно убрать)
    // Serial.printf("Peak: %u\n", peakValue);
  }
}
