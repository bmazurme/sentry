#include <WiFi.h>
#include <PubSubClient.h>
#include <Wire.h>
#include "SSD1306Wire.h"

const char* ssid = "Ent";
const char* password = "";
const char* mqtt_server = "192.168.1.139";
const int mqtt_port = 1883;
const char* client_id = "ESP32_CLIENT";
const char* mqtt_user = "forest";
const char* mqtt_pass = "";

const int noisePin = 32;
const int doPin = 38;

unsigned long lastVibTime = 0;
int maxNoiseValue = 0;
int minNoiseValue = 4095;
const unsigned long sampleWindow = 1000;

int maxVibState = 0;  // максимальное значение вибрации за период

const int OLED_SDA = 4;
const int OLED_SCL = 15;
const int MY_RST_OLED = 16;
const int PIN_VEXT = 21;

SSD1306Wire display(0x3C, OLED_SDA, OLED_SCL);
WiFiClient wifiClient;
PubSubClient client(wifiClient);

bool indicatorActive = false;
unsigned long indicatorStart = 0;
const unsigned long INDICATOR_DURATION = 500;
unsigned long lastDebounceTime = 0;
const unsigned long debounceDelay = 50;
int currentVibState = 1;

void VextON() {
  pinMode(PIN_VEXT, OUTPUT);
  digitalWrite(PIN_VEXT, LOW);
}

void VextOFF() {
  pinMode(PIN_VEXT, OUTPUT);
  digitalWrite(PIN_VEXT, HIGH);
}

void displayReset() {
  pinMode(MY_RST_OLED, OUTPUT);
  digitalWrite(MY_RST_OLED, HIGH);
  delay(1);
  digitalWrite(MY_RST_OLED, LOW);
  delay(1);
  digitalWrite(MY_RST_OLED, HIGH);
  delay(1);
}

void setup_wifi() {
  Serial.println("\nConnecting to WiFi...");
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi connected!");
  Serial.print("IP: ");
  Serial.println(WiFi.localIP());
}

void reconnect() {
  while (!client.connected()) {
    Serial.print("Connecting to MQTT...");
    if (client.connect(client_id, mqtt_user, mqtt_pass)) {
      Serial.println(" connected!");
    } else {
      Serial.print("failed, rc=");
      Serial.print(client.state());
      Serial.println(" retrying in 5s...");
      delay(5000);
    }
  }
}

void updateDisplay(int noisePeak) {
  display.clear();
  display.setTextAlignment(TEXT_ALIGN_LEFT);
  display.setFont(ArialMT_Plain_10);
  display.drawString(0, 0, "IP: " + WiFi.localIP().toString());
  display.drawString(0, 12, "Noise: " + String(noisePeak));
  display.drawString(0, 24, "Vib:   " + String(currentVibState));
  display.setColor(WHITE);
  if (indicatorActive && (millis() - indicatorStart < INDICATOR_DURATION)) {
    display.fillRect(112, 0, 16, 16);
  } else {
    display.drawRect(112, 0, 16, 16);
    indicatorActive = false;
  }
  display.display();
}

void setup() {
  Serial.begin(115200);
  while (!Serial) delay(100);
  VextON();
  displayReset();
  if (!display.init()) {
    Serial.println("OLED init failed!");
    while (1);
  }
  Serial.println("OLED initialized!");
  display.flipScreenVertically();
  display.setFont(ArialMT_Plain_10);

  // Используем INPUT_PULLUP — как в рабочем минимальном коде
  pinMode(doPin, INPUT_PULLUP);

  setup_wifi();
  client.setServer(mqtt_server, mqtt_port);
  updateDisplay(0);
  delay(2000);
}

void loop() {
  static unsigned long lastSendTime = 0;
  const unsigned long sendInterval = 1000;
  unsigned long currentMillis = millis();

  if (!client.connected()) reconnect();
  client.loop();

  if (currentMillis - lastSendTime >= sendInterval) {
    lastSendTime = currentMillis;
    int inMaxNoise = 0, inMinNoise = 4095;
    unsigned long startMillis = currentMillis;

    // ЯВНЫЙ СБРОС ВСЕХ ПЕРЕМЕННЫХ ПЕРЕД ИЗМЕРЕНИЕМ
    maxVibState = 0;
    bool vibrationDetected = false;
    unsigned long lowStartTime = 0;
    unsigned long maxLowDuration = 0;
    bool inLowState = false;
    int lowPulseCount = 0;  // счётчик импульсов LOW

    while (currentMillis - startMillis < sampleWindow) {
      currentMillis = millis();
      int noiseValue = analogRead(noisePin);
      if (noiseValue > inMaxNoise) inMaxNoise = noiseValue;
      if (noiseValue < inMinNoise) inMinNoise = noiseValue;

      int doValue = digitalRead(doPin);
      currentVibState = doValue;

      // ОСНОВНАЯ ЛОГИКА: точное обнаружение длительности LOW
      if (doValue == LOW && !inLowState) {
        // Начало LOW
        inLowState = true;
        lowStartTime = currentMillis;
        lowPulseCount++;  // считаем каждый новый импульс LOW
      } else if (doValue == HIGH && inLowState) {
        // Конец LOW — вычисляем длительность
        unsigned long lowDuration = currentMillis - lowStartTime;
        if (lowDuration > maxLowDuration) {
          maxLowDuration = lowDuration;
        }
        inLowState = false;
      }

      // Антидребезговая обработка для индикатора
      if (doValue == LOW && (currentMillis - lastDebounceTime > debounceDelay)) {
        lastVibTime = currentMillis;
        indicatorActive = true;
        indicatorStart = currentMillis;
        lastDebounceTime = currentMillis;
      }
    }

    // ОБРАБОТКА СЛУЧАЯ, ЕСЛИ ВСЁ ЕЩЁ В LOW В КОНЦЕ ПЕРИОДА
    if (inLowState) {
      unsigned long currentLowDuration = currentMillis - lowStartTime;
      // КРИТИЧЕСКОЕ ИЗМЕНЕНИЕ: ограничиваем максимальную длительность
      // Не позволяем засчитывать весь период как один импульс
      if (currentLowDuration < sampleWindow) {
        if (currentLowDuration > maxLowDuration) {
          maxLowDuration = currentLowDuration;
        }
      } else {
        // Если сигнал был LOW весь период — считаем это ошибкой
        maxLowDuration = 0;
      }
    }

    // УСЛОВИЕ ФИКСАЦИИ ВИБРАЦИИ: минимальная длительность LOW — 80 мс
    // И хотя бы один импульс LOW был обнаружен
    const unsigned long MIN_VIBRATION_DURATION = 80;
    if (maxLowDuration >= MIN_VIBRATION_DURATION && lowPulseCount > 0) {
      vibrationDetected = true;
    } else {
      vibrationDetected = false;
    }

    // ИТОГОВАЯ УСТАНОВКА maxVibState
    if (vibrationDetected) {
      maxVibState = 1;
    } else {
      maxVibState = 0;
    }

    int noisePeak = inMaxNoise - inMinNoise;
    if (noisePeak < 20) noisePeak = 0;

    char noiseStr[10];
    sprintf(noiseStr, "%d", noisePeak);
    char vibMaxStr[10];
    sprintf(vibMaxStr, "%d", maxVibState);

    client.publish("sensor/noise/peak", noiseStr);
    client.publish("sensor/vibration/peak", vibMaxStr);

    Serial.printf("Noise: %d, Max Low Duration: %lu ms, Pulses: %d, Vib Max: %d\n",
              noisePeak, maxLowDuration, lowPulseCount, maxVibState);
    updateDisplay(noisePeak);
  }

  delay(10);
}
