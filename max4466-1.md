#include <WiFi.h>
#include <PubSubClient.h>

// === НАСТРОЙКИ АЦП И ДАТЧИКА ===
const int sampleWindow = 1000;     // Окно сбора данных (мс)
const int noisePin     = 32;      // GPIO 34 — датчик шума (АЦП)
const int aoPin        = 39;      // GPIO 39 для AO датчика SW-18010P
const int doPin        = 38;      // GPIO 38 для DO датчика SW-18010P
const int sampleCount  = 50;

// Настройки WiFi и MQTT (заполните своими данными)
const char* ssid      = "Ent";
const char* password   = "";
const char* mqtt_server = "192.168.50.182";
const int   mqtt_port  = 1883;
const char* client_id  = "ESP32_CLIENT";
const char* mqtt_user  = "ntlstl";
const char* mqtt_pass  = "";


WiFiClient wifiClient;
PubSubClient client(wifiClient);

// Переменные для статистики
unsigned long vibrationCount = 0;   // Вибрации (DO)
unsigned long noiseReads     = 0;     // Считывания шума
unsigned long lastVibTime  = 0;     // Время последней вибрации


int maxNoiseValue          = 0;     // Макс. значение шума
int minNoiseValue          = 4095;  // Мин. значение шума
int maxVibAoValue          = 0;     // Макс. AO вибрации
int minVibAoValue          = 4095;  // Мин. AO вибрации


// === ФУНКЦИИ ===
void setup_wifi() {
  Serial.println("\nConnecting to WiFi...");
  WiFi.begin(ssid, password);
  
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  
  Serial.println("\nWiFi connected!");
  Serial.print("IP: ");
  Serial.println(WiFi.localIP());
}

void reconnect() {
  while (!client.connected()) {
    Serial.print("Connecting to MQTT...");
    if (client.connect(client_id, mqtt_user, mqtt_pass)) {
      Serial.println(" connected!");
    } else {
      Serial.print("failed, rc=");
      Serial.print(client.state());
      Serial.println(" retrying in 5s...");
      delay(5000);
    }
  }
}

void setup() {
  Serial.begin(115200);
  pinMode(doPin, INPUT);  // DO — цифровой вход
  
  setup_wifi();
  client.setServer(mqtt_server, mqtt_port);
}

void loop() {
  static unsigned long lastSendTime = 0;
  const unsigned long sendInterval = 1000;

  if (!client.connected()) reconnect();
  client.loop();

  unsigned long currentTime = millis();
  if (currentTime - lastSendTime >= sendInterval) {
    lastSendTime = currentTime;

    // Сбор данных за sampleWindow мс
    unsigned long startMillis = millis();
    int inMaxNoise = 0, inMinNoise = 4095;
    int inMaxVib = 0, inMinVib = 4095;

    while (millis() - startMillis < sampleWindow) {
      // Шум (MAX4466)
      int noiseValue = analogRead(noisePin);
      if (noiseValue > inMaxNoise) inMaxNoise = noiseValue;
      if (noiseValue < inMinNoise) inMinNoise = noiseValue;

      // Вибрация (SW-18010P)
      int vibAoValue = analogRead(aoPin);
      int doValue = digitalRead(doPin);

      if (vibAoValue > inMaxVib) inMaxVib = vibAoValue;
      if (vibAoValue < inMinVib) inMinVib = vibAoValue;

      if (doValue == LOW) {
        vibrationCount++;
        lastVibTime = millis();
      }
    }

    // Расчёт амплитуд
    int noisePeak = inMaxNoise - inMinNoise;
    int vibPeak = inMaxVib - inMinVib;
    if (noisePeak < 20) noisePeak = 0;
    if (vibPeak < 20) vibPeak = 0;

    // Подготовка данных для MQTT
    char noiseStr[10], vibStr[10], countStr[10], lastStr[10];
    sprintf(noiseStr, "%d", noisePeak);
    sprintf(vibStr,   "%d", vibPeak);
    sprintf(countStr, "%lu", vibrationCount);
    sprintf(lastStr,  "%lu", (millis() - lastVibTime) / 1000);

    // Отправка в MQTT
    client.publish("sensor/noise/peak",   noiseStr);
    client.publish("sensor/vibration/peak", vibStr);
    client.publish("sensor/vibration/count", countStr);
    client.publish("sensor/vibration/last", lastStr);

    // Лог в Serial
    Serial.printf("Noise: %d, Vib: %d, Count: %lu, Last: %lus\n",
                  noisePeak, vibPeak, vibrationCount, (millis() - lastVibTime) / 1000);
  }
}
