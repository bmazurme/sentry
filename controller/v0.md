```c
#include <WiFi.h>
#include <PubSubClient.h>

#include <Wire.h>
#include "SSD1306Wire.h"  // Библиотека для OLED

// === НАСТРОЙКИ WiFi и MQTT ===
const char* ssid       = "";
const char* password   = "";
const char* mqtt_server = "192.168.1.139";
const int   mqtt_port  = 1883;
const char* client_id  = "ESP32_CLIENT";
const char* mqtt_user  = "";
const char* mqtt_pass  = "";



// === ПИНЫ ДАТЧИКОВ ===
const int noisePin = 32;  // АЦП: шум (MAX4466)
const int aoPin    = 39;  // АЦП: вибрация AO (SW-18010P)
const int doPin    = 38;  // Цифровой: вибрация DO (SW-18010P)

// === ПЕРЕМЕННЫЕ ДЛЯ СТАТИСТИКИ ===
unsigned long vibrationCount = 0;
unsigned long lastVibTime  = 0;

int maxNoiseValue = 0;
int minNoiseValue = 4095;
int maxVibAoValue = 0;
int minVibAoValue = 4095;

const unsigned long sampleWindow = 1000;  // Окно сбора данных (мс)


// === ПИНЫ ДИСПЛЕЯ ===
const int OLED_SDA = 4;      // GPIO 4 → SDA
const int OLED_SCL = 15;     // GPIO 15 → SCL
const int RST_OLED = 16;    // GPIO 16 → RST (reset дисплея)
const int PIN_VEXT = 21;   // GPIO 21 → Vext (питание дисплея)

// === ОБЪЯВЛЕНИЕ ДИСПЛЕЯ ===
SSD1306Wire display(0x3C, OLED_SDA, OLED_SCL);  // Адрес I²C

// === КЛИЕНТЫ ===
WiFiClient wifiClient;
PubSubClient client(wifiClient);

// === УПРАВЛЯЮЩИЕ ПЕРЕМЕННЫЕ ИНДИКАТОРА ===
bool indicatorActive = false;
unsigned long indicatorStart = 0;
const unsigned long INDICATOR_DURATION = 500;  // Длительность подсветки (мс)
unsigned long lastDebounceTime = 0;
const unsigned long debounceDelay = 50;  // Антидребезг (мс)

// === ФУНКЦИИ УПРАВЛЕНИЯ ПИТАНИЕМ ДИСПЛЕЯ ===
void VextON() {
  pinMode(PIN_VEXT, OUTPUT);
  digitalWrite(PIN_VEXT, LOW);
}

void VextOFF() {
  pinMode(PIN_VEXT, OUTPUT);
  digitalWrite(PIN_VEXT, HIGH);
}

void displayReset() {
  pinMode(RST_OLED, OUTPUT);
  digitalWrite(RST_OLED, HIGH);
  delay(1);
  digitalWrite(RST_OLED, LOW);
  delay(1);
  digitalWrite(RST_OLED, HIGH);
  delay(1);
}

// === ИНИЦИАЛИЗАЦИЯ WiFi ===
void setup_wifi() {
  Serial.println("\nConnecting to WiFi...");
  WiFi.begin(ssid, password);
  
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  
  Serial.println("\nWiFi connected!");
  Serial.print("IP: ");
  Serial.println(WiFi.localIP());
}

// === ПЕРЕПОДКЛЮЧЕНИЕ MQTT ===
void reconnect() {
  while (!client.connected()) {
    Serial.print("Connecting to MQTT...");
    if (client.connect(client_id, mqtt_user, mqtt_pass)) {
      Serial.println(" connected!");
    } else {
      Serial.print("failed, rc=");
      Serial.print(client.state());
      Serial.println(" retrying in 5s...");
      delay(5000);
    }
  }
}

// === ОБНОВЛЕНИЕ ЭКРАНА ===
void updateDisplay(int noisePeak, int vibPeak, unsigned long vibCount, unsigned long lastVibSecs) {
  display.clear();
  display.setTextAlignment(TEXT_ALIGN_LEFT);
  display.setFont(ArialMT_Plain_10);

  // Вывод IP-адреса
  display.drawString(0, 0, "IP: " + WiFi.localIP().toString());

  // Остальные данные
  display.drawString(0, 12, "Noise: " + String(noisePeak));
  display.drawString(0, 24, "Vib:   " + String(vibPeak));
  display.drawString(0, 36, "Count: " + String(vibCount));
  display.drawString(0, 48, "Last:  " + String(lastVibSecs) + "s");

  // Индикатор активности вибрации
  display.setColor(WHITE);
  if (indicatorActive && (millis() - indicatorStart < INDICATOR_DURATION)) {
    display.fillRect(112, 0, 16, 16);  // Залитый квадрат
  } else {
    display.drawRect(112, 0, 16, 16);    // Контур квадрата
    indicatorActive = false;  // Сбрасываем флаг
  }

  display.display();
}

// === SETUP ===
void setup() {
  Serial.begin(115200);
  while (!Serial) delay(100);

  // Включение питания и сброс дисплея
  VextON();
  displayReset();

  // Инициализация OLED
  if (!display.init()) {
    Serial.println("OLED init failed!");
    while (1);  // Зависаем при ошибке
  }
  Serial.println("OLED initialized!");

  display.flipScreenVertically();
  display.setFont(ArialMT_Plain_10);

  // Настройка пинов датчиков
  pinMode(doPin, INPUT_PULLUP);  // Используем внутренний подтягивающий резистор

  // Подключение к WiFi и MQTT
  setup_wifi();
  client.setServer(mqtt_server, mqtt_port);

  // Первоначальный вывод IP на дисплей
  updateDisplay(0, 0, 0, 0);
  delay(2000);
}

// === LOOP ===
void loop() {
  static unsigned long lastSendTime = 0;
  const unsigned long sendInterval = 1000;
  unsigned long currentMillis = millis();

  if (!client.connected()) reconnect();
  client.loop();

  if (currentMillis - lastSendTime >= sendInterval) {
    lastSendTime = currentMillis;

    // Сброс локальных минимумов/максимумов
    int inMaxNoise = 0, inMinNoise = 4095;
    int inMaxVib = 0, inMinVib = 4095;

    unsigned long startMillis = currentMillis;
    while (currentMillis - startMillis < sampleWindow) {
      currentMillis = millis();

      // Чтение шума
      int noiseValue = analogRead(noisePin);
      if (noiseValue > inMaxNoise) inMaxNoise = noiseValue;
      if (noiseValue < inMinNoise) inMinNoise = noiseValue;

      // Чтение вибрации
      int vibAoValue = analogRead(aoPin);
      int doValue = digitalRead(doPin);

      if (vibAoValue > inMaxVib) inMaxVib = vibAoValue;
      if (vibAoValue < inMinVib) inMinVib = vibAoValue;

      // Обработка цифрового сигнала вибрации с антидребезгом
      if (doValue == LOW && (currentMillis - lastDebounceTime > debounceDelay)) {
        vibrationCount++;
        lastVibTime = currentMillis;
        indicatorActive = true;
        indicatorStart = currentMillis;
        lastDebounceTime = currentMillis;
      }
    }

    // Расчёт пиковых значений
    int noisePeak = inMaxNoise - inMinNoise;
    int vibPeak = inMaxVib - inMinVib;
    if (noisePeak < 20) noisePeak = 0;
    if (vibPeak < 20) vibPeak = 0;
    // Подготовка данных для MQTT
    char noiseStr[10], vibStr[10], countStr[10], lastStr[10];
    sprintf(noiseStr, "%d", noisePeak);
    sprintf(vibStr,   "%d", vibPeak);
    sprintf(countStr, "%lu", vibrationCount);
    sprintf(lastStr,  "%lu", (currentMillis - lastVibTime) / 1000);

    // Отправка в MQTT
    client.publish("sensor/noise/peak",   noiseStr);
    client.publish("sensor/vibration/peak", vibStr);
    client.publish("sensor/vibration/count", countStr);
    client.publish("sensor/vibration/last", lastStr);

    // Лог в Serial
    Serial.printf("Noise: %d, Vib: %d, Count: %lu, Last: %lus\n",
                  noisePeak, vibPeak, vibrationCount, (currentMillis - lastVibTime) / 1000);

    // Обновление дисплея
    updateDisplay(noisePeak, vibPeak, vibrationCount, (currentMillis - lastVibTime) / 1000);
  }

  delay(10);  // Необязательная небольшая задержка для стабильности
}
```
