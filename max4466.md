```c
const int sampleWindow = 50;      // Ширина окна выборки в мс (50 мс = 20 Гц)
const int inputPin = 34;          // Аналоговый вход ESP32 (GPIO 34)
unsigned int adcValue;

void setup() {
  Serial.begin(115200);            // Инициализация Serial на 115200 бод
}

void loop() {
  unsigned long startMillis = millis();  // Начало окна выборки
  unsigned int peakValue = 0;           // Пик-пик амплитуда
  unsigned int inMax = 0;               // Максимальное значение в окне
  unsigned int inMin = 4095;           // Минимальное значение в окне (для ESP32 АЦП = 12 бит → 0–4095)

  // Собираем данные в течение sampleWindow мс
  while (millis() - startMillis < sampleWindow) {
    adcValue = analogRead(inputPin);    // Считываем значение АЦП

    // Отбрасываем аномальные значения (если есть)
    if (adcValue < 4096) {
      if (adcValue > inMax) {
        inMax = adcValue;               // Сохраняем максимум
      }
      else if (adcValue < inMin) {
        inMin = adcValue;              // Сохраняем минимум
      }
    }
  }

  peakValue = inMax - inMin;            // Амплитуда «пик-пик»

  // Фильтруем слишком низкие значения (шум)
  if (peakValue < 20) {
    peakValue = 0;
  }

  // Выводим в консоль
  Serial.println(peakValue);

  delay(100);                           // Задержка между измерениями (можно изменить)
}
```
